function randomIndexes(n,count){
 let s=new Set();
 while(s.size<count) s.add(Math.floor(Math.random()*n));
 return [...s];
}

function showPopup(t){
 popupText.innerText=t;
 popup.style.display="flex";

function closePopup(){
 popup.style.display="none";

// ---------------- LOBBY ----------------

function addAgent(){
 let n=document.getElementById("agentInput").value;
 if(!n)return;
 agents.push({name:n});
 renderAgents();
 document.getElementById("agentInput").value="";
}

function removeAgent(i){
 agents.splice(i,1);
 renderAgents();
}

function renderAgents(){
 agentList.innerHTML="";
 agents.forEach((a,i)=>{
 agentList.innerHTML+=`<li>${a.name} <button class="removeBtn" onclick="removeAgent(${i})">-</button></li>`;
 });
}

function startMission(){
 if(agents.length<3){
  showPopup("Minimum 3 agents");
  return;
 }

 localStorage.setItem("session",JSON.stringify({
  players:agents,
  spyCount:parseInt(spySelect.value),
  time:parseInt(timeSelect.value)
 }));

 newRound();
}

// ---------------- ROUND ----------------

function newRound(){
 const s=JSON.parse(localStorage.getItem("session"));

 agents=s.players.map(p=>({name:p.name}));

 let locNames=Object.keys(locations);
 let loc=locNames[Math.floor(Math.random()*locNames.length)];
 let roles=[...locations[loc]];

 let spies=randomIndexes(agents.length,s.spyCount);

 agents.forEach((a,i)=>{
  a.spy=spies.includes(i);
  if(!a.spy) a.role=roles.pop();
  a.location=loc;
 });

 localStorage.setItem("roundAgents",JSON.stringify(agents));
 localStorage.setItem("roundTime",s.time);

 location="game.html";
}

// ---------------- GAME ----------------

function setupGame(){
 agents=JSON.parse(localStorage.getItem("roundAgents"));
 showTurn();

 hold.onmousedown=hold.ontouchstart=()=>{
 let a=agents[current];
 card.innerHTML=a.spy?"<div class='spy'>YOU ARE THE SPY</div>":`${a.location}<br>${a.role}`;
 };

 hold.onmouseup=hold.ontouchend=()=>{
 card.innerHTML="";
 nextBtn.style.display="block";
 };
}

function showTurn(){
 turn.innerText="PASS PHONE TO "+agents[current].name.toUpperCase();
}

function nextAgent(){
 current++;
 nextBtn.style.display="none";

 if(current>=agents.length){
  startTimerBtn.style.display="block";
  turn.innerText="ALL ROLES DISTRIBUTED";
 } else showTurn();
}

// ---------------- TIMER + ACCUSE ----------------

function startTimer(){
 if(timerStarted)return;
 timerStarted=true;
 startTimerBtn.style.display="none";

 let t=parseInt(localStorage.getItem("roundTime"));

 timerInterval=setInterval(()=>{
 timer.innerText=Math.floor(t/60)+":"+String(t%60).padStart(2,"0");
 t--;
 },1000);

 showAccuse();
}

function showAccuse(){
 accuseArea.innerHTML="<h3>ACCUSATIONS</h3>";
 agents.forEach((a,i)=>{
 accuseArea.innerHTML+=`${a.name} <button onclick="accuse(${i})">Accuse</button><br>`;
 });
}

function accuse(i){
 clearInterval(timerInterval);
 showPopup(agents[i].spy?"MEMBERS WIN!":"SPY WINS!");
}

// ---------------- CONTROLS ----------------

function playAgain(){ newRound(); }

function endGame(){
 localStorage.clear();
 location="index.html";
}

if(location.pathname.includes("game")) setupGame();

function startTimer(){

 timerStarted=true;
 phase="timer";

 startTimerBtn.style.display="none";
 startVoteBtn.style.display="block";
 startVoteBtn.innerText = `START VOTING -- ( ${agents[0].name.toUpperCase()} goes first)`;

 let t=+localStorage.getItem("roundTime");

 timerInterval=setInterval(()=>{
 timer.innerText=Math.floor(t/60)+":"+String(t%60).padStart(2,"0");
 if(t<=0) clearInterval(timerInterval);
 t--;
 },1000);
}

/* ---- VOTING ---- */

function beginVoting(){

 if(phase==="voting")return;
 clearInterval(timerInterval);
 phase="voting";

 voteIndex=0;
 localStorage.setItem("votes",JSON.stringify([]));

 accuseArea.style.display="block";
 startVoteBtn.style.display="none";

 showVote();
}

function showVote(){

 accuseArea.innerHTML=`
 <h3 class="pass glow">${agents[voteIndex].name.toUpperCase()} CAST YOUR VOTE</h3>
 `;

 agents.forEach((a,i)=>{
 accuseArea.innerHTML+=`
 <div class="voteRow">
   <span>${a.name}</span>
   <button class="accuse" onclick="castVote(${i})">VOTE</button>
 </div>`;
 });
}

window.castVote=i=>{

 let v=JSON.parse(localStorage.getItem("votes"));
 v.push(i);
 localStorage.setItem("votes",JSON.stringify(v));

 let last = voteIndex === agents.length-1;

 accuseArea.innerHTML=`
 
 <button class="start" onclick="nextVote()">${last ? "SEE RESULT" : "PASS PHONE TO " + agents[voteIndex + 1].name.toUpperCase()}</button>
 `;
};

window.nextVote=()=>{

 voteIndex++;

 if(voteIndex>=agents.length){
 finalize();
 } else {
 showVote();
 }
};


/* ---- RESULT ---- */

function finalize(){

 phase="finished";

 let votes=JSON.parse(localStorage.getItem("votes"));
 let tally={};

 votes.forEach(v=>tally[v]=(tally[v]||0)+1);

 let max=0,acc=null,tie=false;

 Object.keys(tally).forEach(k=>{
 if(tally[k]>max){max=tally[k];acc=k;tie=false;}
 else if(tally[k]==max) tie=true;
 });

 popupBox.classList.remove("spyTheme","memberTheme");

 if(tie || !agents[acc].spy){
 popupBox.classList.add("spyTheme");
 popupText.innerText="SPY WINS!";
 }else{
 popupBox.classList.add("memberTheme");
 popupText.innerText="MEMBERS WIN!";
 }

 popup.style.display="flex";
 playAgain.style.display="block";
}
